# Determines if a user-provided integer is even or odd
number = int(input("Enter a number: "))

if number % 2 == 0:
    print(f"{number} is Even.")
else:
    print(f"{number} is Odd.")
# Calculates the factorial of a non-negative integer using a recursive function
def factorial(n):
    if n == 0 or n == 1:
        return 1
    return n * factorial(n - 1)

num = 5
print(f"The factorial of {num} is {factorial(num)}.")
